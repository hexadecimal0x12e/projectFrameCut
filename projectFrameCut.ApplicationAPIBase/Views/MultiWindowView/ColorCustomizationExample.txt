MultiWindowView 背景色自定义功能使用示例
===========================================

1. 在 XAML 中自定义颜色
-----------------------

<local:MultiWindowView x:Name="mdiView"
                      ContainerBackgroundColor="#2C2C2C"
                      TaskBarBackgroundColor="#1E1E1E" />

<!-- 创建带自定义颜色的窗口 -->
<Button Text="创建深色窗口" Clicked="CreateDarkWindow" />


2. 在 C# 代码中自定义颜色
-------------------------

// 设置容器背景色
mdiView.ContainerBackgroundColor = Color.FromArgb("#2C2C2C");
mdiView.TaskBarBackgroundColor = Color.FromArgb("#1E1E1E");

// 创建窗口并设置自定义颜色
var window = mdiView.CreateWindow(
    content: new Label { Text = "深色主题窗口", TextColor = Colors.White },
    title: "深色窗口"
);

// 设置窗口颜色
window.WindowBackgroundColor = Color.FromArgb("#1E1E1E");
window.TitleBarBackgroundColor = Color.FromArgb("#252526");
window.TitleBarTextColor = Colors.White;
window.BorderColor = Color.FromArgb("#3F3F46");


3. 预设主题示例
--------------

// 深色主题
void ApplyDarkTheme()
{
    mdiView.ContainerBackgroundColor = Color.FromArgb("#2C2C2C");
    mdiView.TaskBarBackgroundColor = Color.FromArgb("#1E1E1E");
    
    var window = mdiView.CreateWindow(content, "深色窗口");
    window.WindowBackgroundColor = Color.FromArgb("#1E1E1E");
    window.TitleBarBackgroundColor = Color.FromArgb("#252526");
    window.TitleBarTextColor = Colors.White;
    window.BorderColor = Color.FromArgb("#3F3F46");
}

// 浅色主题（默认）
void ApplyLightTheme()
{
    mdiView.ContainerBackgroundColor = Color.FromArgb("#E8E8E8");
    mdiView.TaskBarBackgroundColor = Color.FromArgb("#F5F5F5");
    
    var window = mdiView.CreateWindow(content, "浅色窗口");
    window.WindowBackgroundColor = Colors.White;
    window.TitleBarBackgroundColor = Color.FromArgb("#F0F0F0");
    window.TitleBarTextColor = Colors.Black;
    window.BorderColor = Color.FromArgb("#CCCCCC");
}

4. 完整示例：颜色选择器
---------------------

// 在你的页面中
private void CreateColorCustomWindow()
{
    var stackLayout = new VerticalStackLayout { Padding = 10, Spacing = 10 };
    
    // 窗体背景色选择
    var windowColorPicker = new Button { Text = "选择窗体背景色" };
    windowColorPicker.Clicked += async (s, e) =>
    {
        // 这里可以使用颜色选择器插件
        currentWindow.WindowBackgroundColor = Colors.LightBlue;
    };
    stackLayout.Add(windowColorPicker);
    
    // 标题栏背景色选择
    var titleBarColorPicker = new Button { Text = "选择标题栏颜色" };
    titleBarColorPicker.Clicked += (s, e) =>
    {
        currentWindow.TitleBarBackgroundColor = Colors.DarkBlue;
        currentWindow.TitleBarTextColor = Colors.White;
    };
    stackLayout.Add(titleBarColorPicker);
    
    // 边框颜色选择
    var borderColorPicker = new Button { Text = "选择边框颜色" };
    borderColorPicker.Clicked += (s, e) =>
    {
        currentWindow.BorderColor = Colors.Navy;
    };
    stackLayout.Add(borderColorPicker);
    
    currentWindow = mdiView.CreateWindow(stackLayout, "颜色自定义");
}


5. 可用的颜色属性
----------------

MultiWindowView 属性：
- ContainerBackgroundColor: MDI 容器背景色
- TaskBarBackgroundColor: 任务栏背景色

MultiWindowItem 属性：
- WindowBackgroundColor: 窗体内容区背景色
- TitleBarBackgroundColor: 标题栏背景色
- TitleBarTextColor: 标题栏文字颜色
- BorderColor: 窗口边框颜色

所有颜色属性都支持：
- XAML 绑定
- 运行时动态修改
- 颜色格式：Colors 枚举、#RGB、#RGBA、#RRGGBB、#RRGGBBAA
